# Adapted to the hegemony tier from the custom empire flow in CK3 1.20.0.2.
# Called in the founding character's scope by chg_found_hegemony_decision.
chg_create_custom_hegemony_effect = {
	custom_tooltip = chg_create_title_tt
	custom_tooltip = chg_inherit_identity_tt
	custom_tooltip = chg_held_empires_tt
	custom_tooltip = chg_uncreated_empires_tt
	custom_tooltip = chg_keep_empires_tt

	hidden_effect = {
		save_scope_as = chg_founder
		primary_title = { save_scope_as = old_title }
		# Choose the capital in an existing character scope. Conditional triggers
		# inside the not-yet-created title are also evaluated by tooltip previews.
		if = {
			limit = { exists = scope:old_title.title_capital_county }
			scope:old_title.title_capital_county = { save_scope_as = chg_title_capital }
		}
		else = {
			capital_county = { save_scope_as = chg_title_capital }
		}

		# Snapshot eligible de jure empires before adding or granting any title.
		every_held_title = {
			limit = { chg_eligible_empire_trigger = yes }
			add_to_list = chg_founding_empires
		}
		every_sub_realm_county = {
			empire ?= {
				if = {
					limit = {
						chg_eligible_empire_trigger = yes
						NOT = { exists = holder }
						NOT = { is_in_list = chg_founding_empires }
						# PREV is the empire title while in the founder's scope.
						scope:chg_founder = { completely_controls = prev }
					}
					add_to_list = chg_founding_empires
				}
			}
		}

		# The native name uses old_title.GetBaseNameNoTierNoTooltip.
		create_dynamic_title = {
			tier = hegemony
			name = NEW_CREATED_TITLE_NAME
		}
		scope:new_title = { save_scope_as = chg_new_hegemony }
		create_title_and_vassal_change = {
			type = created
			save_scope_as = chg_title_change
			add_claim_on_loss = no
		}
		scope:chg_new_hegemony = {
			change_title_holder = {
				holder = scope:chg_founder
				change = scope:chg_title_change
			}
		}
		resolve_title_and_vassal_change = scope:chg_title_change

		every_in_list = {
			list = chg_founding_empires
			set_de_jure_liege_title = scope:chg_new_hegemony
		}
		scope:chg_new_hegemony = {
			set_coa = scope:old_title
			set_color_from_title = scope:old_title
			set_capital_county = scope:chg_title_capital
		}
		set_primary_title_to = scope:chg_new_hegemony
		trigger_event = chg.0001
	}
}
